+class L135OptionMenu {
+
+    L135MenuRow[] rows;
+    int pointedRow = 0;
+
+    L135OptionMenu(L135MenuRow... rows) {
+        this.rows = rows;
+    }
+
+    void up() {
+        back();
+        pointedRow = pointedRow - 1;
+        if (pointedRow < 0) {
+            pointedRow = rows.length - 1;
+        }
+    }
+
+    void down() {
+        back();
+        pointedRow = pointedRow + 1;
+        if (pointedRow == rows.length) {
+            pointedRow = 0;
+        }
+    }
+
+    void left() {
+        rows[pointedRow].left();
+    }
+
+    void right() {
+        rows[pointedRow].right();
+    }
+
+    void select() {
+        rows[pointedRow].select();
+    }
+
+    void back() {
+        rows[pointedRow].back();
+    }
+
+    String chosen(String rowName) {
+        L135MenuRow row = rowNamed(rowName);
+        L135MenuItem selectedItem = row.selected();
+        String fullName = selectedItem.fullName();
+        return fullName;
+    }
+
+    void startAt(String rowName, String fullName) {
+        L135MenuRow row = rowNamed(rowName);
+        row.startAt(fullName);
+    }
+
+    String summary() {
+        String summary = "";
+        for (int i = 0; i < rows.length; i++) {
+            L135MenuItem selectedItem = rows[i].selected();
+            if (i > 0) {
+                summary = summary + "-";
+            }
+            summary = summary + selectedItem.shortName();
+        }
+        return summary;
+    }
+
+    String[] lines() {
+        String[] lines = new String[rows.length + 1];
+        lines[0] = summary();
+        for (int i = 0; i < rows.length; i++) {
+            String pointer = "  ";
+            if (i == pointedRow) {
+                pointer = "> ";
+            }
+            lines[i + 1] = pointer + rows[i].line();
+        }
+        return lines;
+    }
+
+    L135MenuRow rowNamed(String rowName) {
+        for (L135MenuRow row : rows) {
+            if (row.name().equals(rowName)) {
+                return row;
+            }
+        }
+        throw new IllegalArgumentException("No row named " + rowName);
+    }

+class L135MenuRow {
+
+    String name;
+    L135MenuItem[] items;
+    int selectedIndex = 0;
+    int offeredIndex = 0;
+
+    L135MenuRow(String name, L135MenuItem... items) {
+        this.name = name;
+        this.items = items;
+    }
+
+    String name() {
+        return name;
+    }
+
+    void right() {
+        int lastIndex = items.length - 1;
+        offeredIndex = offeredIndex + 1;
+        if (offeredIndex > lastIndex) {
+            offeredIndex = 0;
+        }
+    }
+
+    void left() {
+        int lastIndex = items.length - 1;
+        offeredIndex = offeredIndex - 1;
+        if (offeredIndex < 0) {
+            offeredIndex = lastIndex;
+        }
+    }
+
+    void select() {
+        selectedIndex = offeredIndex;
+    }
+
+    void back() {
+        offeredIndex = selectedIndex;
+    }
+
+    L135MenuItem selected() {
+        L135MenuItem selectedItem = items[selectedIndex];
+        return selectedItem;
+    }
+
+    void startAt(String fullName) {
+        for (int i = 0; i < items.length; i++) {
+            if (items[i].fullName().equals(fullName)) {
+                selectedIndex = i;
+                offeredIndex = i;
+            }
+        }
+    }
+
+    String line() {
+        L135MenuItem offeredItem = items[offeredIndex];
+        String choiceShown = "<" + offeredItem.fullName() + ">";
+        if (offeredIndex != selectedIndex) {
+            choiceShown = offeredItem.fullName() + "? A:OK B:Back";
+        }
+        String rowLine = name + ": " + choiceShown;
+        return rowLine;
+    }
+}
